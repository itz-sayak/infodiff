\section{Theory}\label{sec:theory}

\subsection{Exact recursion}
\paragraph{Exact recursion.} Every state in this paper is built from such Erlang chains: the
Hawkes excitation, the response to news, the neural latent state of EPT and the renewal channel of
EPT-X. For all of them, the evolution between two events and its integral are available in closed
form. The following lemma gives the formulas for one source and one rate.

\begin{lemma}[Erlang-chain recursion]\label{lem:recursion}
Let $t_1<t_2<\dots$ be the event times of one source, let $\beta>0$ be one rate of the
dictionary, and define the chain of states
\[
s_r(t)=\sum_{n:\,t_n<t} g_r(t-t_n),\qquad r=0,\dots,R-1,
\]
where $g_r$ is the basis density~(\ref{eq:basis}) with rate $\beta$. Write
$\bm s=(s_0,\dots,s_{R-1})^\top$, let $I$ be the $R\times R$ identity and let $B$ be the
$R\times R$ down-shift matrix ($B_{r,r-1}=1$ and all other entries zero).
\begin{enumerate}
\item Between events, $\dot{\bm s}=\beta(B-I)\,\bm s$; at an event, $s_0$ increases by $\beta$.
\item For a gap $\Delta\ge0$ with no event in $(t,t+\Delta)$, let $y=\beta\Delta$ and
$w_j=e^{-y}y^j/j!$. Then
\[
s_r(t+\Delta)=\sum_{j=0}^{r}w_j\,s_{r-j}(t),\qquad
\int_t^{t+\Delta}s_r(u)\,\mathrm{d}u=\frac{1}{\beta}\sum_{j=0}^{r}P(j+1,y)\,s_{r-j}(t),
\]
where $P(a,y)$ is the regularised lower incomplete gamma function.
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{Step 1 (dynamics).} Differentiating~(\ref{eq:basis}) gives
\[
\frac{\mathrm{d}}{\mathrm{d}t}g_0(t)=-\beta\,g_0(t),\qquad
\frac{\mathrm{d}}{\mathrm{d}t}g_r(t)=\beta\bigl(g_{r-1}(t)-g_r(t)\bigr)\quad(r\ge1).
\]
Summing over past events gives $\dot s_0=-\beta s_0$ and $\dot s_r=\beta(s_{r-1}-s_r)$, which is
$\dot{\bm s}=\beta(B-I)\bm s$. Since $g_r(0)=\beta\,\mathbb{1}[r=0]$, a new event adds $\beta$ to
$s_0$ and leaves the other states unchanged.

\emph{Step 2 (evolution over a gap).} The solution is $\bm s(t+\Delta)=e^{\beta(B-I)\Delta}\bm s(t)$.
The matrices $B$ and $I$ commute and $B^R=0$, so
\[
e^{\beta(B-I)\Delta}=e^{-y}\,e^{yB}=e^{-y}\sum_{j=0}^{R-1}\frac{y^j}{j!}B^j .
\]
Because $B^j$ shifts a vector down by $j$ places, component $r$ of $e^{\beta(B-I)\Delta}\bm s(t)$
is $\sum_{j\le r}w_j s_{r-j}(t)$.

\emph{Step 3 (integral over a gap).} Integrating the weights over the gap,
\[
\int_0^{\Delta}e^{-\beta u}\frac{(\beta u)^j}{j!}\,\mathrm{d}u=\frac{1}{\beta}\,P(j+1,y),
\]
which gives the second formula.
\end{proof}

Evaluating a likelihood therefore costs $O(N d K R^2)$ operations for $N$ events in $d$
dimensions, and it involves no Monte Carlo or quadrature error.

\subsection{Concavity, EM and a certificate of global optimality}
\paragraph{Concavity and a certificate.} Problem~(\ref{eq:ll}) maximises a concave function over
the nonnegative orthant. We add an $\ell_1$ penalty $\bm\gamma^\top\bm\theta_i$ with weights
$\bm\gamma\ge0$ proportional to each feature's exposure; since $\bm\theta_i\ge0$, the penalty is
linear and simply adds $\bm\gamma$ to $\bm\Phi$. We maximise with the multiplicative EM update
\[
\theta_p\;\leftarrow\;\theta_p\,\frac{\sum_n x_{np}/\lambda_n}{\Phi_p+\gamma_p},
\]
where $x_{np}$ is feature $p$ at the $n$-th event of type $i$ and $\lambda_n$ the current intensity
there. We accelerate it with SQUAREM \citep{varadhan2008simple} and finish with a log-barrier
Newton method. The following bound tells us how far any fit is from the optimum.

\begin{proposition}[Duality-gap certificate]\label{prop:gap}
Fix an event type and drop the index $i$. Let $X$ be the matrix whose $n$-th row is
$\bm x(t_n)^\top$ for the $N$ events of this type, let $\bm\theta\ge0$ be any parameter vector with
intensities $\lambda_n=(X\bm\theta)_n>0$, and let $\ell^\star$ be the maximum of the penalised
log-likelihood $\ell(\bm\theta)=\sum_n\log\lambda_n-(\bm\Phi+\bm\gamma)^\top\bm\theta$. Define
\[
s=\max_p\frac{\sum_n x_{np}/\lambda_n}{\Phi_p+\gamma_p}.
\]
Then
\[
\ell^\star-\ell(\bm\theta)\;\le\;N\log s+\bigl(\bm\theta^\top(\bm\Phi+\bm\gamma)-N\bigr),
\]
and both terms on the right are zero at the optimum.
\end{proposition}
\begin{proof}
We use the notation of the proposition: $X$ has rows $\bm x_n^\top=\bm x(t_n)^\top$,
$\lambda_n=\bm x_n^\top\bm\theta$, and
$\ell(\bm\theta)=\sum_{n=1}^{N}\log\lambda_n-(\bm\Phi+\bm\gamma)^\top\bm\theta$.

\emph{Step 1 (an upper bound for every feasible point).} For $u,v>0$, Fenchel's inequality gives
\[
\log u\;\le\;v\,u-1-\log v .
\]
Take any vector $\bm v=(v_1,\dots,v_N)$ with positive entries such that
$X^\top\bm v\le\bm\Phi+\bm\gamma$ componentwise. Applying the inequality with $u=\lambda_n$ and
$v=v_n$ and summing over $n$,
\[
\sum_{n}\log\lambda_n\;\le\;\sum_n v_n\lambda_n-N-\sum_n\log v_n
\;=\;\bm\theta^\top X^\top\bm v-N-\sum_n\log v_n .
\]
Since $\bm\theta\ge0$ and $X^\top\bm v\le\bm\Phi+\bm\gamma$, we have
$\bm\theta^\top X^\top\bm v\le\bm\theta^\top(\bm\Phi+\bm\gamma)$. Subtracting
$(\bm\Phi+\bm\gamma)^\top\bm\theta$ from both sides therefore gives
\[
\ell(\bm\theta)\;\le\;D(\bm v):=-N-\sum_n\log v_n .
\]
This holds for every feasible $\bm\theta$, in particular for the maximiser, so
$\ell^\star\le D(\bm v)$.

\emph{Step 2 (a computable choice of $\bm v$).} Let $\bm\theta$ be the current fit and set
$v_n=1/(s\lambda_n)$, with $s=\max_p\bigl(\sum_n x_{np}/\lambda_n\bigr)/(\Phi_p+\gamma_p)$. Then
$(X^\top\bm v)_p=\frac{1}{s}\sum_n x_{np}/\lambda_n\le\Phi_p+\gamma_p$ by the definition of $s$, so
$\bm v$ is admissible in Step 1. For this choice,
\[
D(\bm v)=-N+N\log s+\sum_n\log\lambda_n ,
\qquad\text{hence}\qquad
\ell^\star-\ell(\bm\theta)\;\le\;D(\bm v)-\ell(\bm\theta)=N\log s+\bm\theta^\top(\bm\Phi+\bm\gamma)-N .
\]

\emph{Step 3 (the bound vanishes at the optimum).} The Karush--Kuhn--Tucker conditions of the
maximisation state that $\sum_n x_{np}/\lambda_n=\Phi_p+\gamma_p$ when $\theta_p>0$ and
$\sum_n x_{np}/\lambda_n\le\Phi_p+\gamma_p$ when $\theta_p=0$. Hence $s=1$, so $N\log s=0$. Moreover,
\[
\bm\theta^\top(\bm\Phi+\bm\gamma)=\sum_p\theta_p\sum_n\frac{x_{np}}{\lambda_n}
=\sum_n\frac{\bm x_n^\top\bm\theta}{\lambda_n}=N ,
\]
so the second term is also zero.
\end{proof}

Every fit reported in this paper comes with this bound.

\subsection{Markov embedding and echo-corrected responses}
\paragraph{Markov embedding.} To study the response to a single release, we collect all
endogenous Erlang states $s_{jkr}$ into one vector $\bm u(t)$ and the states of the release's
news kernels into $\bm v(t)$. Let $F$ be the block-diagonal generator of the endogenous chains
(one block $\beta_k(B-I)$ per source and rate, as in Lemma~\ref{lem:recursion}) and $F_e$ the
generator of the news chains. Let $G_{\mathrm{in}}$ be the matrix that adds $\beta_k$ to the first
phase of every chain of source $j$ when an event of type $j$ occurs. Finally, let $\Theta$ hold the
endogenous weights $a_{ijkr}$ and $\Theta_e$ the news weights of this release, so that the excess
intensity is $\Theta\bm u(t)+\Theta_e\bm v(t)$.

Taking expectations, each event of type $j$ occurs at the mean rate $m_j(t)$ and adds
$G_{\mathrm{in}}\bm e_j$ to $\bm u$. The mean excess intensity $\bm m(t)$ caused by the release
therefore satisfies a linear system:
\begin{equation}\label{eq:embedding}
\dot{\bm z}=\mathcal A\bm z,\qquad
\mathcal A=\begin{pmatrix}F+G_{\mathrm{in}}\Theta & G_{\mathrm{in}}\Theta_e\\ 0 & F_e\end{pmatrix},\qquad
\bm m(t)=C\bm z(t),\qquad C=\begin{pmatrix}\Theta&\Theta_e\end{pmatrix},
\end{equation}
where $\bm z=(\bar{\bm u},\bm v)$ stacks the expected endogenous states and the news states, and
$\bm z(0)=\bm z_0=(\bm 0,\bm v_0)$ with $\bm v_0$ the news states just after the release.

\begin{theorem}[Stability]\label{thm:stab}
Let $G$ be the $d\times d$ branching matrix with entries $G_{ij}=\sum_{k,r}a_{ijkr}$, the expected
number of type-$i$ events triggered directly by one type-$j$ event, and let $\rho(G)$ be its
spectral radius. Then the matrix $F+G_{\mathrm{in}}\Theta$ is Hurwitz (all eigenvalues have negative
real part) if and only if $\rho(G)<1$.
\end{theorem}

\begin{proof}
\emph{Step 1.} Every block $\beta_k(B-I)$ of $F$ is lower triangular with diagonal $-\beta_k<0$
and nonnegative off-diagonal entries. Hence $F$ is Hurwitz with nonnegative off-diagonal entries,
and $-F$ is a nonsingular M-matrix. The matrix $P=G_{\mathrm{in}}\Theta$ is entrywise nonnegative.

\emph{Step 2.} For such a pair, $F+P$ is Hurwitz if and only if $\rho\bigl((-F)^{-1}P\bigr)<1$. This
is the regular-splitting theorem for M-matrices \citep[Thm.~3.29]{varga2000matrix}, applied to
$-(F+P)=(-F)-P$.

\emph{Step 3.} The products $AB$ and $BA$ have the same nonzero eigenvalues, so
\[
\rho\bigl((-F)^{-1}G_{\mathrm{in}}\Theta\bigr)=\rho\bigl(\Theta(-F)^{-1}G_{\mathrm{in}}\bigr).
\]

\emph{Step 4.} Column $j$ of $(-F)^{-1}G_{\mathrm{in}}=\int_0^\infty e^{Ft}G_{\mathrm{in}}\,\mathrm{d}t$
contains, for every chain of source $j$, the integral over time of the state left by a single
type-$j$ event. By Lemma~\ref{lem:recursion} this state is $g_{kr}(t)$, whose integral is one.
Hence
\[
\bigl(\Theta(-F)^{-1}G_{\mathrm{in}}\bigr)_{ij}=\sum_{k,r}a_{ijkr}\int_0^\infty g_{kr}(t)\,\mathrm{d}t
=\sum_{k,r}a_{ijkr}=G_{ij}.
\]
Combining Steps 2 to 4, $F+G_{\mathrm{in}}\Theta$ is Hurwitz if and only if $\rho(G)<1$.
\end{proof}

\begin{corollary}[Closed-form absorption]\label{cor:abs}
Assume $\rho(G)<1$. Then $\mathcal A$ is Hurwitz, because it is block triangular and $F_e$ is
Hurwitz, and the following quantities are available in closed form.
\begin{enumerate}
\item The mean excess intensity after a release, echoes included: $\bm m(t)=Ce^{\mathcal At}\bm z_0$.
\item Its total, the expected number of extra events: $M=\int_0^\infty\bm m(t)\,\mathrm{d}t=-C\mathcal A^{-1}\bm z_0$.
\item The fraction of the response absorbed by time $t$, summed over event types:
\[
\Pi(t)=\frac{\bm 1^\top C\mathcal A^{-1}\bigl(e^{\mathcal At}-I\bigr)\bm z_0}{-\bm 1^\top C\mathcal A^{-1}\bm z_0}.
\]
Its quantiles $t_q=\Pi^{-1}(q)$ are the echo-corrected absorption times; we report $t_{50}$ and $t_{90}$.
\item The amplification $\kappa=M/M_{\mathrm{direct}}$, where $M_{\mathrm{direct}}=-\Theta_eF_e^{-1}\bm v_0$
is the total of the direct news response $\bm m_{\mathrm{direct}}(t)=\Theta_e e^{F_et}\bm v_0$ alone.
\item For an asset whose price is observed through upward and downward crossings of size $\delta$
(types $u$ and $v$), the expected price change up to time $t$:
$\delta\int_0^t\bigl(m_u(s)-m_v(s)\bigr)\,\mathrm{d}s$.
\end{enumerate}
\end{corollary}

The usual ``news half-life'' $\ln 2/\beta$ describes only $\bm m_{\mathrm{direct}}$. When $\rho(G)$
is close to one, the echoes dominate the response, and Section~\ref{sec:results} shows that
the two can differ by one to two orders of magnitude.

\subsection{Closed-form conditional forecasts}
The same system gives forecasts. Suppose a release occurs at time $\tau$, and let
$\mathcal F_\tau$ denote the information available just after it: the history of events, the type
of the release and its surprise. We compute the endogenous states $\bm u(\tau)$ exactly from the
history with Lemma~\ref{lem:recursion}, set the news states $\bm v_0$ from the surprise, and write
$\bm z_0=(\bm u(\tau),\bm v_0)$. Over a short horizon we treat the baseline intensity as a constant
vector $\bm\mu$. Baseline events then also excite the endogenous chains, which adds the constant
input $\bm b=(G_{\mathrm{in}}\bm\mu,\bm 0)$ to the system, so that
$\dot{\bm z}=\mathcal A\bm z+\bm b$. Solving this equation and integrating over $(\tau,\tau+h]$ gives the
expected number of events of each type,
\[
\E\bigl[\bm N(\tau,\tau+h]\,\big|\,\mathcal F_\tau\bigr]
=\bm\mu h+C\Bigl[\mathcal A^{-1}\bigl(e^{\mathcal Ah}-I\bigr)\bm z_0
+\mathcal A^{-1}\bigl(\mathcal A^{-1}(e^{\mathcal Ah}-I)-hI\bigr)\bm b\Bigr].
\]
Because each event corresponds to a price move of size $\delta_a$, this is also a forecast of
realised variance ($\delta_a^2$ times the number of events) and, through the difference between
upward and downward events, of the price drift (Track~D).
